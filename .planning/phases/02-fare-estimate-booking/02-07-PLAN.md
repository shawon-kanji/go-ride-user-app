**`src/features/booking/components/LocationPickerScreen.tsx`**:
```tsx
import { router } from 'expo-router';
import { useState } from 'react';
import { Text, View } from 'react-native';

import { Button } from '../../../components/Button';
import { Card } from '../../../components/Card';
import { FALLBACK_REGION } from '../constants';
import { useCurrentLocation } from '../hooks/useCurrentLocation';
import { useBookingDraftStore } from '../store';
import type { LatLng } from '../types';
import { LocationPickerMap } from './LocationPickerMap';

type Step = 'pickup' | 'dropoff';

/** Rounded placeholder label for a chosen point. No reverse geocoding in v1 —
 *  human-readable addresses are explicitly deferred (02-CONTEXT.md § Deferred Ideas). */
function formatCoordinate(coordinate: LatLng): string {
  return `${coordinate.latitude.toFixed(4)}, ${coordinate.longitude.toFixed(4)}`;
}

/**
 * Sequential map-tap-to-pin location selection (02-CONTEXT.md's locked input method).
 *
 * One screen with internal step state rather than two routes: the two steps have no
 * independent back semantics beyond "go to the other step", and a route split would
 * force lat/lng floats through router params (02-RESEARCH.md § Alternatives Considered).
 *
 * Only the PICKUP step defaults to GPS. The dropoff step starts with no pin by design.
 * A denied permission is silent — the map just opens at FALLBACK_REGION and the rider
 * taps to place the pin themselves. No banner, no crash.
 */
export function LocationPickerScreen() {
  const gps = useCurrentLocation();
  const setPickup = useBookingDraftStore((s) => s.setPickup);
  const setDropoff = useBookingDraftStore((s) => s.setDropoff);

  const [step, setStep] = useState<Step>('pickup');
  const [pickupCoordinate, setPickupCoordinate] = useState<LatLng | null>(null);
  const [dropoffCoordinate, setDropoffCoordinate] = useState<LatLng | null>(null);

  // GPS pre-places the pickup pin exactly once, and only if the rider has not already
  // tapped. Tracked as render-time derived state rather than an effect.
  const [appliedGps, setAppliedGps] = useState(false);
  if (!appliedGps && gps.coordinate) {
    setAppliedGps(true);
    setPickupCoordinate(gps.coordinate);
  }

  const isPickupStep = step === 'pickup';
  const coordinate = isPickupStep ? pickupCoordinate : dropoffCoordinate;
  const initialCenter = gps.coordinate ?? FALLBACK_REGION;

  const handleChange = (next: LatLng) => {
    if (isPickupStep) setPickupCoordinate(next);
    else setDropoffCoordinate(next);
  };

  const handleConfirm = () => {
    if (!coordinate) return;
    if (isPickupStep) {
      setPickup(coordinate);
      setStep('dropoff');
      return;
    }
    setDropoff(coordinate);
    router.push('/(app)/(tabs)/book/quote');
  };

  return (
    <View className="flex-1">
      {/* key forces a remount when the GPS centre first arrives, because
          LocationPickerMap's initialRegion is uncontrolled after mount (plan 02-06). */}
      <LocationPickerMap
        key={`${initialCenter.latitude},${initialCenter.longitude}`}
        initialCenter={initialCenter}
        coordinate={coordinate}
        onCoordinateChange={handleChange}
      />

      <Card className="rounded-none border-x-0 border-b-0">
        <Text className="text-lg font-bold text-neutral-900">
          {isPickupStep ? 'Set pickup' : 'Set dropoff'}
        </Text>
        <Text className="mb-3 mt-1 text-sm text-neutral-600">
          Tap anywhere on the map to place the pin.
        </Text>

        {pickupCoordinate && (
          <Text className="text-sm text-neutral-800">
            Pickup: {formatCoordinate(pickupCoordinate)}
          </Text>
        )}
        {dropoffCoordinate && (
          <Text className="text-sm text-neutral-800">
            Dropoff: {formatCoordinate(dropoffCoordinate)}
          </Text>
        )}

        <View className="mt-4">
          <Button
            label={isPickupStep ? 'Confirm pickup' : 'Confirm dropoff'}
            onPress={handleConfirm}
            disabled={!coordinate}
          />
        </View>

        {!isPickupStep && (
          <View className="mt-2">
            <Button label="Back" variant="ghost" onPress={() => setStep('pickup')} />
          </View>
        )}
      </Card>
    </View>
  );
}
```
Do NOT add a loading spinner gated on `gps.resolving` — `02-RESEARCH.md` Pattern 2 requires the map to render immediately at the fallback and re-centre only if GPS arrives; blocking on `resolving` would make a slow GPS fix feel like a hang.
