**`src/api/cab-client.ts`** — new file, deliberately NOT an extension of `http-client.ts`:
```ts
import type { CabApiErrorBody } from '../features/booking/types';
import { CAB_ERROR_UNKNOWN } from '../features/booking/types';

// A DIFFERENT service on a DIFFERENT port than go-ride-backend (8082 vs 8080).
// Do NOT reuse EXPO_PUBLIC_API_BASE_URL here.
const CAB_BASE_URL = process.env.EXPO_PUBLIC_CAB_API_BASE_URL;

/** cab-request-handler's error body names its sentinel field `error`, not `code`, so
 *  this cannot be go-ride-backend's ApiError. Exposed as `errorCode` to make the
 *  difference impossible to confuse at call sites. */
export class CabApiError extends Error {
  errorCode: string;
  status: number;

  constructor(status: number, body: CabApiErrorBody) {
    super(body.message);
    this.errorCode = body.error;
    this.status = status;
  }
}

interface CabRequestOptions {
  method?: 'GET' | 'POST';
  body?: unknown;
  /** Sent as the Idempotency-Key header (go-ride-utils/httpheaders.Idempotency),
   *  which the server prefers over the body's idempotency_key field. */
  idempotencyKey?: string;
}

/**
 * Purpose-built client for cab-request-handler. Deliberately does NOT reuse
 * src/api/http-client.ts's apiRequest — see 02-RESEARCH.md Pitfall 3:
 *   1. different base URL (port 8082, path prefix /api/v1/cab)
 *   2. error body field is `error`, not `code`
 *   3. this service has NO auth middleware, so a 401 here must never clearSession()
 *      and log the rider out of the whole app over a booking-service hiccup.
 * Do not import useSessionStore into this file.
 */
export async function cabRequest<T>(path: string, options: CabRequestOptions = {}): Promise<T> {
  const { method = 'POST', body, idempotencyKey } = options;

  const headers: Record<string, string> = { 'Content-Type': 'application/json' };
  if (idempotencyKey) headers['Idempotency-Key'] = idempotencyKey;

  const response = await fetch(`${CAB_BASE_URL}${path}`, {
    method,
    headers,
    body: body !== undefined ? JSON.stringify(body) : undefined,
  });

  const json = await response.json().catch(() => ({}));

  if (!response.ok) {
    throw new CabApiError(response.status, {
      error: json.error ?? CAB_ERROR_UNKNOWN,
      message: json.message ?? 'Something went wrong. Please try again.',
    });
  }

  return json as T;
}
```

**`src/features/booking/store.ts`**:
```ts
import { create } from 'zustand';

import type { LatLng } from './types';

interface BookingDraftState {
  pickup: LatLng | null;
  dropoff: LatLng | null;
  setPickup: (coordinate: LatLng | null) => void;
  setDropoff: (coordinate: LatLng | null) => void;
  reset: () => void;
}

/**
 * In-memory draft of the rider's chosen pickup/dropoff, shared between the two booking
 * routes (book/index -> book/quote). No persistence middleware: an unfinished booking
 * draft must not survive an app restart, and this deliberately avoids serialising
 * lat/lng floats through expo-router params.
 *
 * Holds ONLY coordinates. The fare quote itself lives in the quote screen's mutation
 * state so that "Get new estimate" is a plain re-mutate against these same pins.
 */
export const useBookingDraftStore = create<BookingDraftState>((set) => ({
  pickup: null,
  dropoff: null,
  setPickup: (coordinate) => set({ pickup: coordinate }),
  setDropoff: (coordinate) => set({ dropoff: coordinate }),
  reset: () => set({ pickup: null, dropoff: null }),
}));
```
