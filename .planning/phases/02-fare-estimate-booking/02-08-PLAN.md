```tsx
import { router } from 'expo-router';
import { useEffect, useRef, useState } from 'react';
import { ActivityIndicator, Text, View } from 'react-native';

import { CabApiError } from '../../../api/cab-client';
import { Banner } from '../../../components/Banner';
import { Button } from '../../../components/Button';
import { Card } from '../../../components/Card';
import { useFareEstimateMutation, useRequestCabMutation } from '../api';
import { useBookingIdempotencyKey } from '../hooks/useBookingIdempotencyKey';
import { useCountdown } from '../hooks/useCountdown';
import { useRiderId } from '../hooks/useRiderId';
import { useBookingDraftStore } from '../store';
import {
  CAB_ERROR_FARE_ALREADY_USED,
  CAB_ERROR_FARE_EXPIRED,
  CAB_ERROR_FARE_NOT_FOUND,
} from '../types';
import { FareBreakdown } from './FareBreakdown';
import { QuoteCountdown } from './QuoteCountdown';

const GENERIC_RETRY_MESSAGE = 'Something went wrong, please try again.';
const GENERIC_REQUOTE_MESSAGE = 'Something went wrong, please get a new estimate.';
const EXPIRED_MESSAGE = 'Your quote expired.';

export function FareQuoteScreen() {
  const pickup = useBookingDraftStore((s) => s.pickup);
  const dropoff = useBookingDraftStore((s) => s.dropoff);
  const resetDraft = useBookingDraftStore((s) => s.reset);
  const riderId = useRiderId();

  const fareEstimate = useFareEstimateMutation();
  const requestCab = useRequestCabMutation();
  const { getOrCreateKey, resetKey } = useBookingIdempotencyKey();

  const [booked, setBooked] = useState(false);
  const [serverExpired, setServerExpired] = useState(false);
  const [errorMessage, setErrorMessage] = useState<string | null>(null);

  const fare = fareEstimate.data ?? null;

  // expires_at is optional on the wire; absent means "no countdown available", which must
  // not brick the screen. Parsed once per quote rather than on every render tick.
  const expiresAtMs = fare?.expires_at ? new Date(fare.expires_at).getTime() : null;
  const { msRemaining, expired } = useCountdown(expiresAtMs);

  // The countdown reaching zero and the server's 409 fare_expired converge on ONE state
  // (02-CONTEXT.md: "same code path, not a separate error UI").
  const hasExpiry = expiresAtMs !== null;
  const isExpired = hasExpiry && (expired || serverExpired);

  const runEstimate = () => {
    if (!pickup || !dropoff || !riderId) return;
    setErrorMessage(null);
    setServerExpired(false);
    fareEstimate.mutate({
      rider_id: riderId,
      pickup_lat: pickup.latitude,
      pickup_lng: pickup.longitude,
      dropoff_lat: dropoff.latitude,
      dropoff_lng: dropoff.longitude,
    });
  };

  // Fire the first estimate once, when the pins and rider_id are all available.
  // The ref guard is required: riderId can flip from null to a value on a cold start
  // (see plan 02-04's useRiderId), which would otherwise re-fire this effect.
  const estimateStarted = useRef(false);
  useEffect(() => {
    if (estimateStarted.current) return;
    if (!pickup || !dropoff || !riderId) return;
    estimateStarted.current = true;
    runEstimate();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [pickup, dropoff, riderId]);

  const onNewEstimate = () => {
    // A genuinely new booking intent against a new fare — the ONLY place a new
    // idempotency key may be minted (02-CONTEXT.md / RIDE-02).
    resetKey();
    runEstimate();
  };

  const onBookPress = () => {
    if (!fare || !riderId) return;
    setErrorMessage(null);
    requestCab.mutate(
      { rider_id: riderId, fare_id: fare.fare_id, idempotencyKey: getOrCreateKey() },
      {
        onSuccess: () => setBooked(true),
        onError: (error: unknown) => {
          if (error instanceof CabApiError) {
            // A retry that actually succeeded the first time. Idempotent success, not an
            // error — covers both already-booked-earlier and lost-a-concurrent-claim-race.
            if (error.errorCode === CAB_ERROR_FARE_ALREADY_USED) {
              setBooked(true);
              return;
            }
            if (error.errorCode === CAB_ERROR_FARE_EXPIRED) {
              setServerExpired(true);
              setErrorMessage(EXPIRED_MESSAGE);
              return;
            }
            if (error.errorCode === CAB_ERROR_FARE_NOT_FOUND) {
              setErrorMessage(GENERIC_REQUOTE_MESSAGE);
              return;
            }
          }
          setErrorMessage(GENERIC_RETRY_MESSAGE);
        },
      },
    );
  };

  // Success state replaces the screen rather than auto-navigating, mirroring
  // ChangePasswordForm's Phase 1 pattern. Deliberately minimal: NO live "finding driver"
  // status here — that is Phase 3 (TRACK-02) and building it now would be throwaway work.
  if (booked) {
    return (
      <View className="flex-1 px-6 py-6">
        <Banner message="Ride requested!" variant="success" />
        <Text className="mb-4 text-sm text-neutral-600">
          We&apos;ve sent your request. You can head back home while we sort it out.
        </Text>
        <Button
          label="Back to home"
          variant="ghost"
          onPress={() => {
            resetDraft();
            router.replace('/(app)/(tabs)');
          }}
        />
      </View>
    );
  }

  if (!pickup || !dropoff) {
    return (
      <View className="flex-1 px-6 py-6">
        <Banner message="Set a pickup and dropoff first." variant="info" />
        <Button label="Set locations" variant="ghost" onPress={() => router.back()} />
      </View>
    );
  }

  const showRequoteFallback = !isExpired && errorMessage === GENERIC_REQUOTE_MESSAGE;

  return (
    <View className="flex-1 px-6 py-6">
      {errorMessage && <Banner message={errorMessage} variant="error" />}

      {fareEstimate.isPending && <ActivityIndicator testID="fare-estimate-loading" size="large" />}

      {fareEstimate.isError && !fareEstimate.isPending && (
        <View>
          <Banner message={GENERIC_RETRY_MESSAGE} variant="error" />
          <Button label="Try again" onPress={runEstimate} />
        </View>
      )}

      {fare && !fareEstimate.isPending && (
        <Card>
          <Text className="mb-3 text-lg font-bold text-neutral-900">Fare estimate</Text>
          <FareBreakdown fare={fare} />

          <View className="mt-3">
            <QuoteCountdown
              msRemaining={isExpired ? 0 : msRemaining}
              expired={isExpired}
            />
          </View>

          <View className="mt-4">
            {isExpired ? (
              <Button label="Get new estimate" onPress={onNewEstimate} />
            ) : (
              <Button
                label="Book this ride"
                onPress={onBookPress}
                loading={requestCab.isPending}
                disabled={requestCab.isPending || !riderId}
              />
            )}
          </View>

          {showRequoteFallback && (
            <View className="mt-2">
              <Button label="Get new estimate" variant="ghost" onPress={onNewEstimate} />
            </View>
          )}
        </Card>
      )}
    </View>
  );
}
```
