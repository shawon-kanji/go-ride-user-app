```ts
import { useCallback, useRef } from 'react';

import { newIdempotencyKey } from '../../../lib/idempotency';

/**
 * RIDE-02's exactly-once guarantee, client half.
 *
 * TanStack Query v5's useMutation has NO cross-call deduplication — every mutate() is an
 * independent call with its own variables, so a rider's second tap is a brand-new request
 * as far as the library is concerned. `isPending`-disabling the button is good hygiene but
 * is NOT the guard: a fast double-tap before React re-renders, or a force-quit-and-reopen
 * mid-request, both slip past it. The key must therefore live OUTSIDE the mutation, in a
 * ref that survives re-renders and repeated mutate() calls.
 *
 * The server enforces the rest: (rider_id, idempotency_key) replay returns the EXISTING
 * trip request as-is (createOrLoadCabRequest in cab-request-handler), so the client's only
 * job is to keep sending the same key for the same intent.
 *
 * resetKey() is called ONLY on "Get new estimate" after expiry — a genuinely new booking
 * intent against a new fare. Never on a plain retry.
 */
export function useBookingIdempotencyKey() {
  const keyRef = useRef<string | null>(null);

  const getOrCreateKey = useCallback(() => {
    if (!keyRef.current) keyRef.current = newIdempotencyKey();
    return keyRef.current;
  }, []);

  const resetKey = useCallback(() => {
    keyRef.current = null;
  }, []);

  return { getOrCreateKey, resetKey };
}
```

```ts
import { useMutation } from '@tanstack/react-query';

import { cabRequest } from '../../api/cab-client';
import type {
  FareEstimateRequest,
  FareEstimateResponse,
  RequestCabRequest,
  RequestCabResponse,
} from './types';

export const cabApi = {
  /** POST /api/v1/cab/fare-estimate -> 201. Deliberately NO idempotency key:
   *  the endpoint has none server-side and always mints a fresh quote. */
  fareEstimate: (body: FareEstimateRequest) =>
    cabRequest<FareEstimateResponse>('/fare-estimate', { method: 'POST', body }),

  /** POST /api/v1/cab/request-cab -> 202. The key travels as the Idempotency-Key HEADER
   *  only — the server prefers the header over the body field (firstNonEmpty in server.go),
   *  so sending both would be redundant and could diverge. */
  requestCab: (body: RequestCabRequest, idempotencyKey: string) =>
    cabRequest<RequestCabResponse>('/request-cab', { method: 'POST', body, idempotencyKey }),
};

export function useFareEstimateMutation() {
  return useMutation({
    mutationFn: (variables: FareEstimateRequest) => cabApi.fareEstimate(variables),
  });
}

/** Variables carry the key alongside the body; the mutationFn splits them so the key
 *  can only ever become a header. No onSuccess cache writes: a fare quote is a
 *  short-lived, single-consumer value owned by the quote screen, not shared server state. */
export type RequestCabVariables = RequestCabRequest & { idempotencyKey: string };

export function useRequestCabMutation() {
  return useMutation({
    mutationFn: ({ idempotencyKey, ...body }: RequestCabVariables) =>
      cabApi.requestCab(body, idempotencyKey),
  });
}
```
Do NOT add a `retry` option to either mutation — `createTestQueryClient()` sets `retry: false` for tests, and in production the idempotency key (not automatic retries) is the duplicate guard. Do NOT add `onSuccess` cache writes.

**Why this hook exists — a real defect `02-CONTEXT.md` did not anticipate.** `02-CONTEXT.md` states `rider_id` comes from the Zustand session store's `User.id`, and that is correct immediately after login/signup because `setSession(token, user)` writes it. But `hydrate()` — the cold-start path — sets only `{ status, token, tokenExpiresAt }` and deliberately leaves `user` null, with the comment "user snapshot is refetched by the profile query once authenticated". So on any app restart, `useSessionStore.getState().user` is `null`, and a naive read would post `rider_id: undefined`, which the server rejects (`rider_id must be a valid UUID`, per `validateFareEstimateRequest`). This hook closes that gap by falling back to the same `GET /me` query the profile screen already uses — no new endpoint, no JWT decoding (`src/lib/jwt.ts` only exposes `decodeJwtExpiryMs` and must not be extended).

```ts
import { useProfileQuery } from '../../profile/api';
import { useSessionStore } from '../../../stores/session-store';

/**
 * Resolves rider_id for POST /fare-estimate and POST /request-cab.
 *
 * Primary source is the session store's User.id, populated by setSession() at
 * login/signup. But session-store.hydrate() (the cold-start path) restores ONLY the
 * token — it never restores `user` — so after any app restart that value is null.
 * Falling back to the already-cached GET /me profile query covers that case without
 * a new endpoint and without decoding the JWT client-side (src/lib/jwt.ts exposes
 * only decodeJwtExpiryMs by design; do not extend it).
 *
 * Returns null while neither source has resolved. Callers MUST gate on null rather
 * than sending rider_id: undefined — the server validates rider_id as a UUID.
 */
export function useRiderId(): string | null {
  const sessionUserId = useSessionStore((state) => state.user?.id ?? null);
  const profile = useProfileQuery();
  return sessionUserId ?? profile.data?.user.id ?? null;
}
```
