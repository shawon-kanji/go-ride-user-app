```ts
import { useEffect, useState } from 'react';
import * as Location from 'expo-location';

import type { LatLng } from '../types';

export interface CurrentLocationResult {
  /** The device position, or null if unavailable for ANY reason. */
  coordinate: LatLng | null;
  /** True until the permission+position sequence settles, either way. */
  resolving: boolean;
}

/**
 * One-shot foreground GPS read for the pickup screen's default pin (02-CONTEXT.md).
 *
 * Deliberately exposes NO error channel. A denied permission or a disabled location
 * service is not an error condition in this flow — 02-CONTEXT.md and 02-RESEARCH.md's
 * Anti-Patterns section both require a silent fallback to manual pin placement, with no
 * banner and no crash. Every failure mode collapses to `coordinate: null`, which the
 * screen treats identically to "GPS not available yet".
 *
 * The caller must NOT block rendering on `resolving` — show the map immediately at the
 * fallback region and re-centre if and when a coordinate arrives.
 */
export function useCurrentLocation(): CurrentLocationResult {
  const [coordinate, setCoordinate] = useState<LatLng | null>(null);
  const [resolving, setResolving] = useState(true);

  useEffect(() => {
    let cancelled = false;

    (async () => {
      try {
        const permission = await Location.requestForegroundPermissionsAsync();
        // Defensive optional-chain: jest-expo's stock auto-mock resolves `undefined`
        // here (02-RESEARCH.md Pitfall 4), and a destructure would throw.
        if (permission?.status !== 'granted') return;

        const position = await Location.getCurrentPositionAsync({});
        if (cancelled || !position?.coords) return;
        setCoordinate({
          latitude: position.coords.latitude,
          longitude: position.coords.longitude,
        });
      } catch {
        // Location services disabled device-wide, or any other native failure.
        // Same graceful path as a denial — the rider places the pin manually.
      } finally {
        if (!cancelled) setResolving(false);
      }
    })();

    return () => {
      cancelled = true;
    };
  }, []);

  return { coordinate, resolving };
}
```

```tsx
import MapView, { Marker, PROVIDER_GOOGLE } from 'react-native-maps';

import { MAP_DELTA } from '../constants';
import type { LatLng } from '../types';

interface LocationPickerMapProps {
  /** Where the viewport opens. Applied ONCE on mount — see the initialRegion note below. */
  initialCenter: LatLng;
  /** The currently placed pin, or null for "nothing placed yet". Fully controlled. */
  coordinate: LatLng | null;
  onCoordinateChange: (coordinate: LatLng) => void;
}

/**
 * Tap-to-pin map (02-CONTEXT.md's locked location-input method).
 *
 * Uses `initialRegion`, NEVER `region`. `region` is a fully controlled prop: every parent
 * re-render that supplies it forces the viewport, which fights the rider's own pan and
 * zoom — and since the pin's coordinate lives in parent state, that re-render happens on
 * every single tap. `initialRegion` is uncontrolled after mount (same semantics as a text
 * input's defaultValue), so the map owns its viewport thereafter.
 * See 02-RESEARCH.md Pattern 1 and its Anti-Patterns section.
 *
 * The map's own onPress is the entire interaction — no draggable marker, no long-press.
 * react-native-maps gives `event.nativeEvent.coordinate` as `{latitude, longitude}`
 * directly; only those two fields are forwarded upward.
 */
export function LocationPickerMap({
  initialCenter,
  coordinate,
  onCoordinateChange,
}: LocationPickerMapProps) {
  return (
    <MapView
      testID="picker-map"
      provider={PROVIDER_GOOGLE}
      style={{ flex: 1 }}
      initialRegion={{ ...initialCenter, ...MAP_DELTA }}
      onPress={(event) => {
        const { latitude, longitude } = event.nativeEvent.coordinate;
        onCoordinateChange({ latitude, longitude });
      }}
    >
      {coordinate && <Marker testID="picker-marker" coordinate={coordinate} />}
    </MapView>
  );
}
```
Note: `initialCenter` is a mount-time value. The pickup screen (plan 02-07) must therefore not render this component until it has decided its centre — it will use a `key` change or a mount gate to re-centre once GPS resolves. That decision belongs to the screen, not here.
