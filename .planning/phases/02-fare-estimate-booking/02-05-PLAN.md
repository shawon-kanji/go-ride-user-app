**`src/features/booking/hooks/useCountdown.ts`**:
```ts
import { useEffect, useState } from 'react';

const TICK_MS = 1000;

export interface CountdownResult {
  /** Milliseconds left, clamped at 0. null when there is no target. */
  msRemaining: number | null;
  expired: boolean;
}

/**
 * Anchored-timestamp countdown, adapted from SessionExpiryBanner's proven shape
 * (see 02-RESEARCH.md Pattern 3) with three deliberate differences:
 *   - 1-second tick, not 30 — this drives the quote screen's primary MM:SS display
 *   - the anchor is the fare quote's expires_at, not a decoded JWT exp claim
 *   - it reports an explicit `expired` flag so the caller can render a terminal state
 *
 * NEVER decrement a local counter: that drifts under any JS-thread pause (app
 * backgrounded, GC, debugger) and never self-corrects. Recomputing `targetMs - Date.now()`
 * from an absolute anchor every tick self-corrects on the first tick after foregrounding,
 * which matters here because the quote window is 15 minutes — plenty of time for a rider
 * to background the app and come back.
 */
export function useCountdown(targetMs: number | null): CountdownResult {
  const [now, setNow] = useState(() => Date.now());

  useEffect(() => {
    if (targetMs === null) return;
    setNow(Date.now());
    const interval = setInterval(() => setNow(Date.now()), TICK_MS);
    return () => clearInterval(interval);
  }, [targetMs]);

  if (targetMs === null) return { msRemaining: null, expired: false };

  const msRemaining = Math.max(0, targetMs - now);
  return { msRemaining, expired: msRemaining <= 0 };
}

/** Milliseconds -> "MM:SS", clamped at 00:00. Minutes are not capped at 59 —
 *  a quote is 15 minutes, but a longer TTL should read as "20:00", not "00:00". */
export function formatMmSs(ms: number): string {
  const totalSeconds = Math.max(0, Math.floor(ms / 1000));
  const minutes = Math.floor(totalSeconds / 60);
  const seconds = totalSeconds % 60;
  return `${String(minutes).padStart(2, '0')}:${String(seconds).padStart(2, '0')}`;
}
```
Note the `setNow(Date.now())` inside the effect before starting the interval: it re-anchors immediately when `targetMs` changes (Test 7) instead of waiting a full second.

import { Text } from 'react-native';

import { formatMmSs } from '../hooks/useCountdown';

const URGENT_THRESHOLD_MS = 60 * 1000;

interface QuoteCountdownProps {
  /** From useCountdown(expiresAtMs). null means no expiry is known — render nothing. */
  msRemaining: number | null;
  expired: boolean;
}

/** Presentational only — the quote screen owns the useCountdown instance and passes
 *  its result down, so the Book button and this label can never disagree by a tick. */
export function QuoteCountdown({ msRemaining, expired }: QuoteCountdownProps) {
  if (msRemaining === null) return null;

  if (expired) {
    return (
      <Text testID="quote-countdown-expired" className="text-sm font-semibold text-danger-700">
        Quote expired
      </Text>
    );
  }

  const urgent = msRemaining <= URGENT_THRESHOLD_MS;

  return (
    <Text
      testID={urgent ? 'quote-countdown-urgent' : 'quote-countdown'}
      className={`text-sm font-semibold ${urgent ? 'text-warning-700' : 'text-neutral-700'}`}
    >
      Quote expires in {formatMmSs(msRemaining)}
    </Text>
  );
}

import { Text, View } from 'react-native';

import type { FareEstimateResponse } from '../types';

function formatAmount(currencyCode: string, amount: number): string {
  return `${currencyCode} ${amount.toFixed(2)}`;
}

function Row({ label, value, testID }: { label: string; value: string; testID?: string }) {
  return (
    <View testID={testID} className="flex-row items-center justify-between py-1.5">
      <Text className="text-sm text-neutral-600">{label}</Text>
      <Text className="text-sm text-neutral-900">{value}</Text>
    </View>
  );
}

/**
 * Itemized fare rows (RIDE-01). Render rules are locked in 02-CONTEXT.md:
 *  - base/distance/time/total: always
 *  - surcharge/discount: ONLY when nonzero (both are always 0 with today's backend
 *    config; a permanent "USD 0.00 discount" row would imply a promo system that
 *    does not exist — PROJECT.md Out of Scope)
 *  - surge_multiplier: NEVER, under any value (hardcoded 1.0 server-side,
 *    PROJECT.md Out of Scope excludes all surge UI)
 */
export function FareBreakdown({ fare }: { fare: FareEstimateResponse }) {
  const money = (amount: number) => formatAmount(fare.currency_code, amount);

  return (
    <View>
      <Row label="Base fare" value={money(fare.base_fare)} />
      <Row label="Distance" value={money(fare.distance_fare)} />
      <Row label="Time" value={money(fare.time_fare)} />
      {fare.surcharge_total !== 0 && <Row label="Surcharge" value={money(fare.surcharge_total)} />}
      {fare.discount_total !== 0 && <Row label="Discount" value={money(fare.discount_total)} />}

      <View testID="fare-total" className="mt-2 flex-row items-center justify-between border-t border-neutral-200 pt-3">
        <Text className="text-base font-semibold text-neutral-900">Total</Text>
        <Text className="text-xl font-bold text-neutral-900">{money(fare.total_fare)}</Text>
      </View>
    </View>
  );
}
